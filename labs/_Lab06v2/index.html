    1. Work-Sharing Directives

    Work-sharing directives distribute tasks among threads within a parallel region. The #pragma omp for directive is your primary tool for parallelizing loops, dividing iterations across threads for efficient computation.

    1.1 The for Directive

    The #pragma omp for directive splits loop iterations among threads. It must be used within a parallel region and requires:

    • A determinable loop trip count.
    • An integer or pointer-type loop variable.
    • Unchanging loop bounds during execution.

    Syntax:

    #pragma omp for [clauses]
for (init; condition; increment)

    Example 1.1: Basic Parallel For

    This example demonstrates separate parallel and for directives.

    #include <iostream>
#include <omp.h>

int main() {
    const int n = 16;
    int arr[n];

    std::cout << "=== Method 1: Separate parallel and for directives ===\n";
    #pragma omp parallel num_threads(4)
    {
        #pragma omp for
        for (int i = 0; i < n; i++) {
            arr[i] = 2 * i + 1;
            std::cout << "Thread " << omp_get_thread_num()
                      << " processing i = " << i
                      << " arr[" << i << "] = " << arr[i] << "\n";
        }
    }
    return 0;
}

    Example 1.2: Combined Parallel For

    This example uses the combined parallel for directive for brevity.

    #include <iostream>
#include <omp.h>

int main() {
    const int n = 16;
    int arr[n];

    std::cout << "=== Method 2: Combined parallel for directive ===\n";
    #pragma omp parallel for num_threads(4)
    for (int i = 0; i < n; i++) {
        arr[i] = 2 * i + 1;
        std::cout << "Thread " << omp_get_thread_num()
                  << " processing i = " << i
                  << " arr[" << i << "] = " << arr[i] << "\n";
    }
    return 0;
}

    1.2 Loop-Carried Dependencies

    Loops with dependencies between iterations (loop-carried dependencies) cannot be safely parallelized without special handling, as iterations may depend on previous results.

    Example 1.3: Fibonacci Sequence - Incorrect Parallelization

    This example shows why parallelizing a Fibonacci sequence computation fails due to dependencies.

    #include <iostream>
#include <omp.h>

int main() {
    const int n = 10;
    long fibo[n];

    // Serial version (correct)
    std::cout << "=== Serial Fibonacci ===\n";
    fibo[0] = fibo[1] = 1;
    for (int i = 2; i < n; i++) {
        fibo[i] = fibo[i-1] + fibo[i-2];
    }
    std::cout << "Serial result: ";
    for (int i = 0; i < n; i++) {
        std::cout << fibo[i] << " ";
    }
    std::cout << "\n";

    // Parallel version (incorrect - has loop-carried dependency)
    std::cout << "=== Parallel Fibonacci (INCORRECT) ===\n";
    fibo[0] = fibo[1] = 1;
    #pragma omp parallel for num_threads(4)
    for (int i = 2; i < n; i++) {
        fibo[i] = fibo[i-1] + fibo[i-2]; // Race condition!
        std::cout << "Thread " << omp_get_thread_num()
                  << " computed fibo[" << i << "] = " << fibo[i] << "\n";
    }
    std::cout << "Parallel result: ";
    for (int i = 0; i < n; i++) {
        std::cout << fibo[i] << " ";
    }
    std::cout << "\n";
    return 0;
}

    2. Tutorial 1: Estimating π with the Leibniz Algorithm

    The Leibniz formula for π is a series: \(\pi/4 = 1 - 1/3 + 1/5 - 1/7 + \cdots\). We'll compute this sum in serial and parallel versions, addressing synchronization challenges.

    2.1 Serial π Estimation

    This example computes π using the Leibniz series sequentially.

    #include <iostream>
#include <chrono>
#include <cmath>

int main() {
    long n = 100000000; // Number of terms
    double sum = 0.0;

    auto start = std::chrono::high_resolution_clock::now();
    for (long i = 0; i < n; i++) {
        sum += (i % 2 == 0 ? 1.0 : -1.0) / (2.0 * i + 1);
    }
    double pi_estimate = 4.0 * sum;

    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);

    std::cout << "Serial π estimation\n";
    std::cout << "Estimated π = " << pi_estimate << "\n";
    std::cout << "Actual π = " << M_PI << "\n";
    std::cout << "Error = " << std::fabs(pi_estimate - M_PI) << "\n";
    std::cout << "Time = " << duration.count() << " ms\n";
    return 0;
}

    2.2 Parallel π Estimation - Incorrect Implementation

    This version has a race condition due to shared access to sum.

    #include <iostream>
#include <omp.h>
#include <chrono>
#include <cmath>

int main() {
    long n = 100000000;
    double sum = 0.0;

    auto start = std::chrono::high_resolution_clock::now();
    #pragma omp parallel for num_threads(4)
    for (long i = 0; i < n; i++) {
        sum += (i % 2 == 0 ? 1.0 : -1.0) / (2.0 * i + 1); // Race condition!
    }
    double pi_estimate = 4.0 * sum;

    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);

    std::cout << "Parallel π estimation (INCORRECT)\n";
    std::cout << "Estimated π = " << pi_estimate << "\n";
    std::cout << "Actual π = " << M_PI << "\n";
    std::cout << "Error = " << std::fabs(pi_estimate - M_PI) << "\n";
    std::cout << "Time = " << duration.count() << " ms\n";
    return 0;
}

    2.3 Parallel π Estimation - Correct Implementation with Reduction

    Using the reduction clause avoids race conditions.

    #include <iostream>
#include <omp.h>
#include <chrono>
#include <cmath>

int main() {
    long n = 100000000;
    double sum = 0.0;

    auto start = std::chrono::high_resolution_clock::now();
    #pragma omp parallel for num_threads(4) reduction(+:sum)
    for (long i = 0; i < n; i++) {
        sum += (i % 2 == 0 ? 1.0 : -1.0) / (2.0 * i + 1);
    }
    double pi_estimate = 4.0 * sum;

    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);

    std::cout << "Parallel π estimation (CORRECT)\n";
    std::cout << "Estimated π = " << pi_estimate << "\n";
    std::cout << "Actual π = " << M_PI << "\n";
    std::cout << "Error = " << std::fabs(pi_estimate - M_PI) << "\n";
    std::cout << "Time = " << duration.count() << " ms\n";
    return 0;
}

    2.4 Parallel π Estimation - Using Private Variables and Critical Section

    This alternative uses private variables and a critical section for synchronization.

    #include <iostream>
#include <omp.h>
#include <chrono>
#include <cmath>

int main() {
    long n = 100000000;
    double sum = 0.0;

    auto start = std::chrono::high_resolution_clock::now();
    #pragma omp parallel num_threads(4)
    {
        double local_sum = 0.0;
        #pragma omp for
        for (long i = 0; i < n; i++) {
            local_sum += (i % 2 == 0 ? 1.0 : -1.0) / (2.0 * i + 1);
        }
        #pragma omp critical
        sum += local_sum;
    }
    double pi_estimate = 4.0 * sum;

    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);

    std::cout << "Parallel π estimation with critical section\n";
    std::cout << "Estimated π = " << pi_estimate << "\n";
    std::cout << "Actual π = " << M_PI << "\n";
    std::cout << "Error = " << std::fabs(pi_estimate - M_PI) << "\n";
    std::cout << "Time = " << duration.count() << " ms\n";
    return 0;
}

    3. Loop Scheduling

    OpenMP offers scheduling strategies to control how loop iterations are distributed:

    • static: Iterations are divided evenly before execution.
    • dynamic: Iterations are assigned as threads become available.
    • guided: Chunk sizes decrease over time.
    • runtime: Schedule set via environment variable.

    Example 3.1: Comparing Scheduling Strategies

    This example demonstrates the impact of scheduling on performance.

    #include <iostream>
#include <omp.h>
#include <chrono>
#include <cmath>

double compute_function(int i) {
    double result = 0.0;
    for (int j = 0; j < i * 1000; j++) { // Variable workload
        result += std::sin(j * 0.1);
    }
    return result;
}

void test_schedule(const char* schedule_name, int n, int num_threads) {
    std::cout << "Testing " << schedule_name << " schedule\n";
    double sum = 0.0;
    auto start = std::chrono::high_resolution_clock::now();

    if (std::string(schedule_name) == "static") {
        #pragma omp parallel for num_threads(num_threads) schedule(static)
        for (int i = 0; i < n; i++) {
            sum += compute_function(i);
            std::cout << "Thread " << omp_get_thread_num() << " processed iteration " << i << "\n";
        }
    } else if (std::string(schedule_name) == "dynamic") {
        #pragma omp parallel for num_threads(num_threads) schedule(dynamic, 2)
        for (int i = 0; i < n; i++) {
            sum += compute_function(i);
            std::cout << "Thread " << omp_get_thread_num() << " processed iteration " << i << "\n";
        }
    } else if (std::string(schedule_name) == "guided") {
        #pragma omp parallel for num_threads(num_threads) schedule(guided)
        for (int i = 0; i < n; i++) {
            sum += compute_function(i);
            std::cout << "Thread " << omp_get_thread_num() << " processed iteration " << i << "\n";
        }
    }

    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);
    std::cout << "Time for " << schedule_name << " schedule: " << duration.count() << " ms\n";
}

int main() {
    const int n = 20;
    const int num_threads = 4;
    std::cout << "Comparing scheduling strategies with " << num_threads << " threads and " << n << " iterations\n";
    test_schedule("static", n, num_threads);
    test_schedule("dynamic", n, num_threads);
    test_schedule("guided", n, num_threads);
    return 0;
}

    4. The Single Directive

    The #pragma omp single directive ensures a block is executed by only one thread, with others waiting at an implicit barrier.

    Syntax:

    #pragma omp single [clauses]
structured block

    Example 4.1: Basic Single Directive

    This example shows single-threaded execution within a parallel region.

    #include <iostream>
#include <omp.h>

int main() {
    std::cout << "Before parallel region\n";
    #pragma omp parallel num_threads(4)
    {
        std::cout << "Thread " << omp_get_thread_num() << " before single region\n";
        #pragma omp single
        {
            std::cout << "Only thread " << omp_get_thread_num() << " executes this single block\n";
            for (int i = 0; i < 3; i++) {
                std::cout << "Single thread working... step " << i + 1 << "\n";
            }
        }
        std::cout << "Thread " << omp_get_thread_num() << " after single region\n";
    }
    return 0;
}

    Example 4.2: Single Directive with Copyprivate

    The copyprivate clause broadcasts a variable's value to all threads.

    #include <iostream>
#include <omp.h>

int main() {
    int x = 5;
    std::cout << "Initial x = " << x << "\n";
    #pragma omp parallel num_threads(4) default(none) private(x)
    {
        std::cout << "Thread " << omp_get_thread_num() << " before single: x = " << x << " (uninitialized)\n";
        #pragma omp single copyprivate(x)
        {
            x = 42;
            std::cout << "Thread " << omp_get_thread_num() << " in single block set x = " << x << "\n";
        }
        std::cout << "Thread " << omp_get_thread_num() << " after single: x = " << x << "\n";
    }
    std::cout << "After parallel region: x = " << x << "\n";
    return 0;
}

    5. Tutorial 2: Bubble Sort and Odd-Even Transposition Sort

    Sorting algorithms provide a great context for exploring parallelization challenges. We'll compare serial and parallel sorting methods.

    5.1 Serial Bubble Sort

    This is a straightforward bubble sort implementation.

    #include <iostream>
#include <vector>
#include <chrono>
#include <random>
#include <algorithm>

void bubble_sort_serial(std::vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                std::swap(arr[j], arr[j + 1]);
            }
        }
    }
}

int main() {
    const int n = 10000;
    std::vector<int> arr(n);
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_int_distribution<> dis(1, 1000);
    for (int& val : arr) {
        val = dis(gen);
    }

    auto start = std::chrono::high_resolution_clock::now();
    bubble_sort_serial(arr);
    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);

    std::cout << "Serial bubble sort time: " << duration.count() << " ms\n";
    bool is_sorted = std::is_sorted(arr.begin(), arr.end());
    std::cout << "Array is sorted: " << (is_sorted ? "Yes" : "No") << "\n";
    return 0;
}

    5.2 Serial Odd-Even Transposition Sort

    This algorithm alternates comparisons between odd and even indices, making it more parallelizable.

    #include <iostream>
#include <vector>
#include <chrono>
#include <random>
#include <algorithm>

void odd_even_sort_serial(std::vector<int>& arr) {
    int n = arr.size();
    bool sorted = false;
    while (!sorted) {
        sorted = true;
        // Odd phase
        for (int i = 1; i < n - 1; i += 2) {
            if (arr[i] > arr[i + 1]) {
                std::swap(arr[i], arr[i + 1]);
                sorted = false;
            }
        }
        // Even phase
        for (int i = 0; i < n - 1; i += 2) {
            if (arr[i] > arr[i + 1]) {
                std::swap(arr[i], arr[i + 1]);
                sorted = false;
            }
        }
    }
}

int main() {
    const int n = 10000;
    std::vector<int> arr(n);
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_int_distribution<> dis(1, 1000);
    for (int& val : arr) {
        val = dis(gen);
    }

    auto start = std::chrono::high_resolution_clock::now();
    odd_even_sort_serial(arr);
    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);

    std::cout << "Serial odd-even sort time: " << duration.count() << " ms\n";
    bool is_sorted = std::is_sorted(arr.begin(), arr.end());
    std::cout << "Array is sorted: " << (is_sorted ? "Yes" : "No") << "\n";
    return 0;
}

    5.3 Parallel Odd-Even Transposition Sort

    This version parallelizes the odd and even phases.

    #include <iostream>
#include <vector>
#include <chrono>
#include <random>
#include <algorithm>
#include <omp.h>

void odd_even_sort_parallel(std::vector<int>& arr, int num_threads) {
    int n = arr.size();
    for (int phase = 0; phase < n; phase++) {
        if (phase % 2 == 1) {
            #pragma omp parallel for num_threads(num_threads)
            for (int i = 1; i < n - 1; i += 2) {
                if (arr[i] > arr[i + 1]) {
                    std::swap(arr[i], arr[i + 1]);
                }
            }
        } else {
            #pragma omp parallel for num_threads(num_threads)
            for (int i = 0; i < n - 1; i += 2) {
                if (arr[i] > arr[i + 1]) {
                    std::swap(arr[i], arr[i + 1]);
                }
            }
        }
    }
}

int main() {
    const int n = 10000;
    const int num_threads = 4;
    std::vector<int> arr(n);
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_int_distribution<> dis(1, 1000);
    for (int& val : arr) {
        val = dis(gen);
    }

    auto start = std::chrono::high_resolution_clock::now();
    odd_even_sort_parallel(arr, num_threads);
    auto end = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(end - start);

    std::cout << "Parallel odd-even sort time: " << duration.count() << " ms\n";
    bool is_sorted = std::is_sorted(arr.begin(), arr.end());
    std::cout << "Array is sorted: " << (is_sorted ? "Yes" : "No") << "\n";
    return 0;
}
